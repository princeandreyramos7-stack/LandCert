⚙️ HOSTINGER - CONFIGURATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
☐ Copy .env.example to .env
☐ Edit .env file and update:
   ☐ APP_ENV=production
   ☐ APP_DEBUG=false
   ☐ APP_URL=https://yourdomain.com
   ☐ DB_HOST=localhost
   ☐ DB_DATABASE=(your database name)
   ☐ DB_USERNAME=(your database user)
   ☐ DB_PASSWORD=(your database password)
   ☐ MAIL_HOST=smtp.hostinger.com
   ☐ MAIL_USERNAME=(your email)
   ☐ MAIL_PASSWORD=(your email password)
   ☐ SEMAPHORE_API_KEY=(your SMS API key)


🔐 HOSTINGER - PERMISSIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
☐ Set permissions (via File Manager or SSH):
   ☐ storage/ → 755
   ☐ bootstrap/cache/ → 755
   ☐ .env → 644


🔑 HOSTINGER - GENERATE APP KEY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
☐ Option 1 (SSH): php artisan key:generate
☐ Option 2 (Manual):
   - Visit: https://generate-random.org/laravel-key-generator
   - Copy generated key
   - Add to .env: APP_KEY=base64:your_key_here


💾 HOSTINGER - DATABASE MIGRATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
☐ Via SSH (if available):
   php artisan migrate --force
   php artisan db:seed --class=RoleSeeder
   php artisan db:seed --class=AdminUserSeeder

☐ Without SSH:
   Create public/migrate.php with migration script
   Visit: https://yourdomain.com/migrate.php
   DELETE migrate.php after use!

☐ Verify in phpMyAdmin:
   Check tables exist: users, requests, payments, certificates


🔗 HOSTINGER - STORAGE LINK
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
☐ Via SSH: php artisan storage:link
☐ Without SSH: Create symlink via File Manager
☐ Verify: public/storage folder exists


🧹 HOSTINGER - CACHE & OPTIMIZE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

   php artisan cache:clear
   php artisan config:clear
   php artisan route:clear
   php artisan view:clear

☐ Cache for production:
   php artisan config:cache
   php artisan route:cache
   php artisan view:cache
